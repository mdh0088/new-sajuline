## 1. 테이블 생성 DDL

### 1.1 천간 테이블 (saju_cheongan)

```sql
CREATE TABLE IF NOT EXISTS `saju_cheongan` (
    `id` INT NOT NULL COMMENT '천간 ID (1~10)',
    `name` VARCHAR(2) NOT NULL COMMENT '천간 이름 (갑~계)',
    `hanja` VARCHAR(4) NOT NULL COMMENT '한자 (甲~癸)',
    `oheng` VARCHAR(4) NOT NULL COMMENT '오행 (목, 화, 토, 금, 수)',
    `yin_yang` VARCHAR(4) NOT NULL COMMENT '음양 (양, 음)',
    `description` TEXT NOT NULL COMMENT '천간 설명 및 특성',
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='천간 (天干) 기초 데이터';
```

### 1.2 지지 테이블 (saju_jiji)

```sql
CREATE TABLE IF NOT EXISTS `saju_jiji` (
    `id` INT NOT NULL COMMENT '지지 ID (1~12)',
    `name` VARCHAR(2) NOT NULL COMMENT '지지 이름 (자~해)',
    `hanja` VARCHAR(4) NOT NULL COMMENT '한자 (子~亥)',
    `oheng` VARCHAR(4) NOT NULL COMMENT '오행 (목, 화, 토, 금, 수)',
    `yin_yang` VARCHAR(4) NOT NULL COMMENT '음양 (양, 음)',
    `animal` VARCHAR(4) NOT NULL COMMENT '12지신 동물',
    `month` INT NOT NULL COMMENT '대응 음력월 (자=11월)',
    `description` TEXT NOT NULL COMMENT '지지 설명 및 특성',
    PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='지지 (地支) 기초 데이터';
```

### 1.3 십성 해석 테이블 (saju_sipsung_interpretation)

```sql
CREATE TABLE IF NOT EXISTS `saju_sipsung_interpretation` (
    `id` INT NOT NULL AUTO_INCREMENT COMMENT '해석 ID',
    `ilgan` VARCHAR(2) NOT NULL COMMENT '일간 (갑~계)',
    `sipsung` VARCHAR(10) NOT NULL COMMENT '십성',
    `keywords` JSON NOT NULL COMMENT '핵심 키워드 배열',
    `positive_meaning` TEXT NOT NULL COMMENT '긍정적 의미 해석',
    `negative_meaning` TEXT NOT NULL COMMENT '부정적 의미 해석',
    `daily_advice` TEXT NOT NULL COMMENT '일일 운세 조언',
    PRIMARY KEY (`id`),
    INDEX `idx_ilgan_sipsung` (`ilgan`, `sipsung`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='십성 해석 데이터';
```

### 1.4 지지 관계 테이블 (saju_jiji_relation)

```sql
CREATE TABLE IF NOT EXISTS `saju_jiji_relation` (
    `id` INT NOT NULL AUTO_INCREMENT COMMENT '관계 ID',
    `relation_type` VARCHAR(10) NOT NULL COMMENT '관계 유형 (합, 충, 형, 파, 해)',
    `jiji1` VARCHAR(2) NOT NULL COMMENT '첫 번째 지지',
    `jiji2` VARCHAR(2) NOT NULL COMMENT '두 번째 지지',
    `name` VARCHAR(20) NOT NULL COMMENT '관계 이름',
    `keywords` JSON NOT NULL COMMENT '핵심 키워드 배열',
    `meaning` TEXT NOT NULL COMMENT '관계 의미 해석',
    `daily_impact` TEXT NOT NULL COMMENT '일일 운세에 미치는 영향',
    PRIMARY KEY (`id`),
    INDEX `idx_relation` (`relation_type`, `jiji1`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='지지 관계 데이터';
```

### 1.5 운세 이력 테이블 (fortune_histories)

```sql
CREATE TABLE IF NOT EXISTS `fortune_histories` (
    `id` CHAR(36) NOT NULL COMMENT '운세 이력 ID (UUID)',
    `user_id` VARCHAR(100) NOT NULL COMMENT '사용자 ID',
    `fortune_type` VARCHAR(20) NOT NULL COMMENT '운세 유형 (daily, weekly, monthly, yearly)',
    `target_date` DATE NOT NULL COMMENT '운세 대상 날짜',
    `content` TEXT NOT NULL COMMENT 'AI 생성 운세 내용',
    `ai_model` VARCHAR(50) NOT NULL COMMENT '사용된 AI 모델',
    `prompt_version` VARCHAR(20) NOT NULL COMMENT '프롬프트 버전',
    `created_at` DATETIME NOT NULL COMMENT '생성일시',
    PRIMARY KEY (`id`),
    INDEX `idx_user_type_date` (`user_id`, `fortune_type`, `target_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='AI 운세 생성 이력';
```

SELECT COUNT(*) AS cheongan_count FROM saju_cheongan;
